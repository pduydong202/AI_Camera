cmake_minimum_required(VERSION 3.16)
project(motion_filter_l123_tawl_wsl LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
 set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
find_package(OpenCV REQUIRED COMPONENTS core imgproc video videoio imgcodecs)
add_executable(motion_filter src/main.cpp src/config.cpp src/motion_pipeline.cpp src/event_manager.cpp src/offline_recorder.cpp src/validation_recorder.cpp src/full_mask_recorder.cpp)
target_include_directories(motion_filter PRIVATE include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(motion_filter PRIVATE ${OpenCV_LIBS} pthread)
target_compile_options(motion_filter PRIVATE -Wall -Wextra -Wpedantic $<$<CONFIG:Release>:-O3>)
